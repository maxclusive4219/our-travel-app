Google Sheets Sync Configuration

ผูกหน้าแอปนี้เข้ากับ Google Sheet ส่วนตัวของคุณเพื่อใช้งานออนไลน์แบบถาวรฟรี 100%

วางลิงก์ Web App ที่คุณได้จากการบันทึก Google Apps Script ที่ระบบสร้างให้ลงในนี้ เพื่อเริ่มการเชื่อมต่อแบบออนไลน์

Google Apps Script Code สำหรับก๊อปปี้ วางใน Extensions > Apps Script 
function doGet() {
  var userProperties = PropertiesService.getUserProperties();
  var data = userProperties.getProperty('trip_database');
  return ContentService.createTextOutput(data || '{"plans": ["Plan A", "Plan B"], "tripData": {"A":[], "B":[]}, "expenses": []}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var jsonString = e.postData.contents;
    var userProperties = PropertiesService.getUserProperties();
    userProperties.setProperty('trip_database', jsonString);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
